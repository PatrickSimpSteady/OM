function getCards() {

  return document.querySelectorAll('.question-card');

}


function showQuestion(index) {

  getCards().forEach(c => c.classList.remove('active'));

  getCards()[index].classList.add('active');

  currentQ = index;

  updateProgress();

  restoreSelection(index);

}


function updateProgress() {

  const pct = Math.round(((currentQ) / questions) * 100);

  document.getElementById('progress-bar').style.width = pct + '%';

  document.getElementById('progress-label').textContent = 'Question ' + (currentQ + 1) + ' of ' + questions;

}


function restoreSelection(index) {

  const card = getCards()[index];

  const btns = card.querySelectorAll('.option-btn');

  const nextBtn = document.getElementById('next-' + index);

  btns.forEach(b => b.classList.remove('selected'));

  if (answers[index] !== null) {

    btns[answers[index]].classList.add('selected');

    if (nextBtn) { nextBtn.classList.add('enabled'); }

  } else {

    if (nextBtn) { nextBtn.classList.remove('enabled'); }

  }

}


document.querySelectorAll('.question-card').forEach((card, qIndex) => {

  const btns = card.querySelectorAll('.option-btn');

  const nextBtn = document.getElementById('next-' + qIndex);

  const backBtn = document.getElementById('back-' + qIndex);


  btns.forEach((btn, bIndex) => {

    btn.addEventListener('click', () => {

      btns.forEach(b => b.classList.remove('selected'));

      btn.classList.add('selected');

      answers[qIndex] = bIndex;

      if (nextBtn) nextBtn.classList.add('enabled');

    });

  });


  if (nextBtn) {

    nextBtn.addEventListener('click', () => {

      if (answers[qIndex] === null) return;

      if (qIndex === questions - 1) {

        showResults();

      } else {

        showQuestion(qIndex + 1);

      }

    });

  }


  if (backBtn) {

    backBtn.addEventListener('click', () => {

      showQuestion(qIndex - 1);

    });

  }

});


function showResults() {

  let r1 = 0, r2 = 0, r3 = 0;

  answers.forEach(a => {

    if (a === 0) r1++;

    else if (a === 1) r2++;

    else if (a === 2) r3++;

  });


  const total = r1 + r2 + r3;

  const p1 = Math.round((r1 / total) * 100);

  const p2 = Math.round((r2 / total) * 100);

  const p3 = Math.round((r3 / total) * 100);


  let mode;

  if (r1 >= r2 && r1 >= r3) mode = 'reactive';

  else if (r2 >= r1 && r2 >= r3) mode = 'transitional';

  else mode = 'structural';


  const data = resultData[mode];


  document.getElementById('quiz-section').style.display = 'none';

  document.getElementById('results-section').style.display = 'block';


  document.getElementById('result-badge').textContent = data.badge;

  document.getElementById('result-badge').className = 'result-mode-badge ' + data.badgeClass;

  document.getElementById('result-title').textContent = data.title;

  document.getElementById('result-score-line').textContent =

    r1 + ' Reactive · ' + r2 + ' Transitional · ' + r3 + ' Structural answers';

  document.getElementById('result-description').innerHTML = data.description;

  document.getElementById('result-next').innerHTML = data.next;


  setTimeout(() => {

    document.getElementById('bar-reactive').style.width = p1 + '%';

    document.getElementById('bar-transitional').style.width = p2 + '%';

    document.getElementById('bar-structural').style.width = p3 + '%';

    document.getElementById('pct-reactive').textContent = p1 + '%';

    document.getElementById('pct-transitional').textContent = p2 + '%';

    document.getElementById('pct-structural').textContent = p3 + '%';

  }, 100);


  window.scrollTo({ top: 0, behavior: 'smooth' });

}


function retakeQuiz() {

  answers.fill(null);

  document.querySelectorAll('.option-btn').forEach(b => b.classList.remove('selected'));

  document.querySelectorAll('.btn-next').forEach(b => b.classList.remove('enabled'));

  document.getElementById('results-section').style.display = 'none';

  document.getElementById('quiz-section').style.display = 'block';

  showQuestion(0);

  window.scrollTo({ top: 0, behavior: 'smooth' });

}
